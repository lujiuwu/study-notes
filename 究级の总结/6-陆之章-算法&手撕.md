
### 防抖&节流

- 防抖： 在一段时间内，事件只会最后触发一次
- 节流： 事件按照一段时间的间隔来进行触发

**防抖**
```js
function debounce(fn,wait,immediate = false){
    let timeout = null
    const _debounce = function(){
      // 上一个定时器未结束 -- 重置
      if(timeout) clearTimeout(timeout)
      // 是否立即执行
      if(immediate){
        // 如果上一个定时器已结束 -- 这次立即执行
        let isNow = !timeout
        // wait期间timeout不为null -- 确保不会重复执行fn
        timeout = setTimout(()=>{
          timeout = null
        },wait)
        // 再执行函数
        // 如果位置调换，wait内就可能执行多次fn
        if(isNow) fn.apply(this,arguments)
      }
      // 到时执行 -- 赋值 证明timeout被触发
      else timeout = setTimeout(()=>{
        fn.apply(this,arguments)
      },wait)
    }
    // 取消功能
    _debounce.cancel = function(){
      clearTimeout(timeout)
      timeout = null
    }
    return _debounce
}
```

**节流**
```js
function throttle(fn,wait){
  let timeout
  return function(){
    if(!timeout){
      timeout = setTimeout(()=>{
        timeout = null
        fn.apply(this,arguments)
      },wait)
    }
  }
}
```

### instanceof( v1, v2 )
**原理**
检测v2(构造函数)的 _prototype_ 属性是否出现在实例对象v1的原型链上
* v1必须是对象
* v2必须是函数
```js
function _instanceof(obj,fn){
  if(typeof obj !== 'object' || obj === null) return false
  if(typeof fn !== 'function') throw new TypeError('val2 must be function')
  // 正常判断
  let proto = obj.__proto__
  while(proto){
    if(proto === fn.proptotype) return true
    proto = proto.__proto__
  }
  return false

}
```

### 手写map

```js
Array.prototype.map = function(callback, thisArg) {
  if (this == undefined) {
    throw new TypeError('this is null or not defined');
  }
  if (typeof callback !== 'function') {
    throw new TypeError(callback + ' is not a function');
  }
  const res = [];
  // 同理
  const O = Object(this);
  const len = O.length >>> 0;
  for (let i = 0; i < len; i++) {
    if (i in O) {
      // 调用回调函数并传入新数组
      res[i] = callback.call(thisArg, O[i], i, this);
    }
  }
  return res;
}

```
### Promise

**手写Promise**
```js
const PENDING = 'pending'
const FULFILLED = 'fulfilled'
const REJECTED = 'rejected'
class MyPromise{
  constructor(executor){
    // 初始化状态
    this.status = PENDING
    // 初始化事件数组
    this.fulfilledFnList = []
    this.rejectedFnList = []
    // 初始化接收值
    this.value = null
    this.reason = null
  
    // 构建 resolve 与 reject
    const resolve = (value = unll)=>{
      if(this.status === PENDING){
        this.status = FULFILLED
        this.value = value
        fulfilledFnList.forEach(fn=>{
          // 微任务
          queueMicrotask(()=>fn(this.value))
        })
      }
    }
    const reject = (reason = unll)=>{
      if(this.status === PENDING){
        this.status = FULFILLED
        this.reason = reason
        rejectedFnList.forEach(fn=>{
          // 微任务
          queueMicrotask(()=>fn(this.reason))
        })
      }
    }
    // 立即执行
    executor(resolve,reject)
  }
  // then方法
  _then(onResolved,onRejected){
    onResolved&&this.fulfilledFnList.push(onResolved)
    onRejected&&this.rejectedFnList.push(onRejected)
  }
  _catch(onRejected){
    this._then(null,onRejected)
  }
  _finally(finallyFn){
    this._then(()=>{
      finallyFn()
    },()=>{
      finallyFn()
    })
  }
}
```

* all -- 全部成功或返回第一个rejected
```js
static function _all(promises){
  return new Promise((resolve,reject)=>{
    const arr = []
    promises.forEach(promise=>{
      promise.then(res=>{
        arr.push(res)
        if(arr.length === promises.length) resolve(arr)
      },err=>{
        reject(err)
      })
    })
  })
}
```

* allSettled -- 返回所有执行结果数组，结构为{status:'',value/reason}
```js
static function _allSettled(promises){
  return new Promise((resolve,reject)=>{
    const arr = []
    promises.forEach(promise=>{
      promise.then(res=>{
        arr.push({status:'fulfilled',value:res})
        if(arr.length === promises.length) resolve(arr)
      },err=>{
        arr.push({status:'rejected',reason:err})
        if(arr.length === promises.length) resolve(arr)
      })
    })
  })
  
}
```

* rece -- 返回第一个执行完成的结果
```js
static function _rece(promises){

    })
  })
}
```

```js
function _apply(context=window,args=[]){
  // 避免属性冲突
  const fn = Symbol('fn')
  context[fn] = this
  // 执行函数
  const res = context[fn](...args)
  delete context[fn];
  return res
}
```

### 深拷贝

```js
function _clone(target,map = new WeakMap()){
  // 如果是对象才进入递归逻辑
  if(typeof target === 'object' && target !== null){
    // 是否已经复制过
    if(map.get(target)) return map.get(target)
    // 基础结构
    const cloneTarget = Array.isArray(target)?[]:{}
    for(const key in target){
      cloneTarget[key] = clone(target[key],map)
    }
    return cloneTarget
  }
  // 不是则直接返回
  else return target
}
```
