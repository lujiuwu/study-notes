function foo(root){

### 二叉树的深度

**高度与深度的区别**

* 深度 -- 二叉树任一节点到根节点的距离 -- 前序遍历
* 高度 -- 二叉树任一节点到叶子节点的距离 -- 后序遍历


**111 - 最小深度**

```js
var minDepth = function(root) {
    if(!root) return 0
    const minLeftDepth = minDepth(root.left)
    const minRightDepth = minDepth(root.right)

    if(!root.right && root.left) return minLeftDepth+1
    else if(!root.left && root.right) return minRightDepth+1
    else return Math.min(minLeftDepth,minRightDepth)+1

};
```

function maxDepth(root){

### 完全二叉树

**222-完全二叉树节点个数**

* 层级遍历
```js
var countNodes = function(root) {

    if(!root) return 0

  

    let current = [root]

    let res = 1

  

    while(current.length){

        const next = []

        for(const node of current){

            if(!node.right) return node.left ? res+1:res

            else {

                res+=2

                next.push(node.left)

                next.push(node.right)

            }

        }

        current = next

    }

    return res

};
```

* 后序遍历

```js
var countNodes = function(root) {

   // 终止条件

   if(!root) return 0

   if(!root.left) return 1

  

   // 处理逻辑

   const leftNodes = countNodes(root.left)

   const rightNodes = countNodes(root.right)

  

   const totalNodes = leftNodes+rightNodes+1

   return totalNodes

  

};
```

### 平衡二叉树

**110**
* 任一节点的左右子节点的高度差不得大于1（可以等于1）

```js
function isBalanced(root){
  function getDepth(root){
    if(!root) return 0
    const lDepth = getDepth(root.left)
    if(lDepth == -1) return -1
    const rDepth = getDepth(root.right)
    if(rDepth == -1) return -1
    if(Math.abs(lDepth-rDepth)>1) return -1
    else return 1+Math.max(lDepth,rDepth)
  }
  return getDepth(root)!==-1
}
```
