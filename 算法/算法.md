/**

 * @param {number[]} nums

 * @return {number}

 */

var longestConsecutive = function(nums) {

    let ans = 0

    const hashSet = new Set(nums)

    for(const item of hashSet){

        if(hashSet.has(item-1)) continue;

        else {

          let last = item+1

          while(hashSet.has(last)){last++}

          ans = Math.max(ans,last-item)

        }

    }

    return ans

};

### 三数之和

```js
/**

 * @param {number[]} nums

 * @return {number[][]}

 */

var threeSum = function(nums) {

   let ans = []

   nums.sort((a,b)=>a-b)

   for(let i = 0;i<nums.length;i++){

    if(nums[i]>0) break;

    if(i>0&&nums[i]===nums[i-1]) continue

    let r = nums.length-1,l = i+1

    while(l<r){

        const sum = nums[i]+nums[r]+nums[l]

        if(sum === 0){

            ans.push([nums[i],nums[r],nums[l]])

            // 去重

            while(l<r&&nums[l]===nums[l+1]) l++

            while(r>l&&nums[r]===nums[r-1]) r--

            l++

            r--

        }

        else if(sum<0) l++

        else r--

    }

   }

   return ans

};
```

var hasCycle = function(head) {

    const set = new Set()

    let headNode = head

    while(headNode){

        const nextNode = headNode.next

        if(!nextNode) return false

        if(set.has(nextNode)){

            return true

        }else{

            set.add(nextNode)

        }

        headNode = headNode.next

    }

    return false

};

var mergeTwoLists = function(l1, l2) {

  // 链表为空的情况

  if(!l1) return l2

  if(!l2) return l1

  if(l1.val<l2.val){

    l1.next = mergeTwoLists(l1.next,l2)

    return l1

  }

  else{

    l2.next = mergeTwoLists(l2.next,l1)

    return l2
