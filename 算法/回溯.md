
**递归&回溯**

* 回溯搜索是纯暴力的算法
* 可以解决的问题：
     1. 组合问题
     2. 切割问题
     3. 子集问题
     4. 排列问题
     5. 棋盘问题

**如何理解**
* 回溯可以抽象成树形结构 -- 一个n叉树
* 树的宽度就是回溯中处理的集合大小
* 树的深度就是递归的深度

**回溯法的模板**
* 递归函数（backtracking）无返回值
* 参数
* 终止条件 if( )
  1. 收集结果
  2. return
* 单层搜索逻辑
  1. for循环 -- 参数用于处理集合的元素集；处理节点；参与递归
  2. 递归完成后 -- 回溯操作；撤销处理节点的操作
  3. 结束for循环
  4. 正常return

#### leetcode 77

```js
var combine = function(n, k) {

  const res = []

  function back_tracking(arr,num){

    // 终止条件

    // 如果当前的数组长度arr.length === k --> 一次收集结束

    // 1. res.push(arr)

    if(arr.length === k){

        res.push([...arr])

        return

    }

    // 正常处理逻辑

    for(let i = num;i<=n;i++){

        // 更新arr

        arr.push(i)

        // 递归

        back_tracking(arr,i+1)

        // 回溯

        arr.pop()

    }

    return

  }

  back_tracking([],1)

  return res

};
```

```js
/**

 * @param {number} k

 * @param {number} n

 * @return {number[][]}

 */

var combinationSum3 = function(k, n) {

    const res = []

    function backTracking(arr,sum,startNum){

        // 终止条件 -- 长度为k个且和为n

        if(arr.length === k && sum === n){

            res.push([...arr])

            return

        }

        // 正常逻辑

        for(let i = startNum;i<=9;i++){

            arr.push(i)

            sum+=i

            backTracking(arr,sum,i+1)

            // 回溯

            arr.pop()

            sum-=i

        }

        return

    }

    backTracking([],0,1)

    return res

};
```


#### 17 电话号码的字母组合

```js
/**

 * @param {string} digits

 * @return {string[]}

 */

  

// 需要初始化映射吧？

// 形如：2 -> ['a','b','c']

const map = new Map()

map.set('2',['a','b','c'])

map.set('3',['d','e','f'])

map.set('4',['g','h','i'])

map.set('5',['j','k','l'])

map.set('6',['m','n','o'])

map.set('7',['p','q','r','s'])

map.set('8',['t','u','v'])

map.set('9', ['w', 'x', 'y', 'z'])

var letterCombinations = function(digits) {

  if(digits === '') return []

  if(digits.length === 1) return map.get(digits)

   const res = []

   // 递归函数的参数：

   // 1. strs -- 数组 存储时转换类型

   // 2. digitIndex -- 当前层级的digits字符串索引数

   function backTracking(strs,digitIndex){

    // 终止条件 -- 递归到digits末尾

    if(digitIndex === digits.length){

        res.push(strs.join(''))

        return

    }

    const arr = map.get(digits[digitIndex])

    for(let i = 0;i<arr.length;i++){

        strs.push(arr[i])

        backTracking(strs,digitIndex+1)

        strs.pop()

    }

    return

   }

   backTracking([],0)

   return res

};
```
